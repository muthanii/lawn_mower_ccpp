% CAS's abstract takes an optional title it never uses; looking ahead for it reads
% the abstract's first command before the verbatim copy starts, and that command is
% lost (here the first \textbf). Drop the optional argument.
\RenewDocumentEnvironment{abstract}{}{\verbatimwrite{\jobname.abs}}{\endverbatimwrite}

% CAS sets the keywords box beside the abstract, then backs up by the box's height
% without reserving it. When the abstract is shorter than the keyword list, the
% keywords run into the first section. Pad the abstract to the keywords' height.
\makeatletter
\newdimen\paper@keyht
\newsavebox\paper@absbox
\AddToHook{env/Abstract/begin}{%
  \global\paper@keyht=\dimexpr\ht\csname g_stm_key_box\endcsname
                             +\dp\csname g_stm_key_box\endcsname\relax
  \global\sbox\paper@absbox{\vbox{\hsize=.65\textwidth \parindent\z@
    \noindent\abstractname\par\vskip-4pt
    \noindent\rule{.65\textwidth}{.2pt}\par
    \footnotesize\everypar{\parindent1.5em}\noindent\ignorespaces
    \IfFileExists{\jobname.abs}{\input{\jobname.abs}}{}\par}}}
\AddToHook{env/Abstract/end}{%
  \par
  \ifdim\dimexpr\ht\paper@absbox+\dp\paper@absbox\relax<\paper@keyht
    \vskip\dimexpr\paper@keyht-\ht\paper@absbox-\dp\paper@absbox\relax
  \fi}
\makeatother

% CAS writes the page count for its "Page x of y" footer before pending floats are
% flushed, so trailing table or figure pages are missed. Take it from lastpage.
\usepackage{lastpage}
\AtBeginDocument{\def\lastpage{\pageref*{LastPage}}}

% CAS sets the corresponding-author, email and ORCID notes as footnotes on the title
% page. Whenever a float is met, LaTeX's output routine puts the page back and
% re-inserts pending footnotes after it, at the float's position. If the page then
% breaks above that position, the title-page notes move to the next page. Re-insert
% them before the page instead, so they stay where they were set. \unvcopy keeps
% \footins full for the kernel's room check; the later \@reinserts skip it.
\usepackage{etoolbox}
\makeatletter
\def\paper@reinsertsnofn{\global\setbox\footins\box\voidb@x \@reinserts}
\patchcmd\@specialoutput{\unvbox \@holdpg \@next}
  {\ifvoid\footins\else\insert\footins{\unvcopy\footins}\fi \unvbox \@holdpg \@next}{}
  {\PackageWarningNoLine{main}{Could not keep the title-page notes on page 1}}
\patchcmd\@specialoutput{\@reinserts \@addtocurcol}{\paper@reinsertsnofn \@addtocurcol}{}
  {\PackageWarningNoLine{main}{Could not keep the title-page notes on page 1}}
\patchcmd\@specialoutput{\@reinserts \@addmarginpar}{\paper@reinsertsnofn \@addmarginpar}{}
  {\PackageWarningNoLine{main}{Could not keep the title-page notes on page 1}}
\makeatother
